    Dim entryIds As Variant
    Dim index As Long

    entryIds = Split(EntryIDCollection, ",")
    For index = LBound(entryIds) To UBound(entryIds)
        HiworksRulesFinal.ClassifyIncomingMail CStr(entryIds(index))

End Sub
